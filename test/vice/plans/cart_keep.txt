# The cartridge with a drive and a disk in it (run.sh with SETDISK): the settings are read from and kept on
# the disk, so the settings screen does not say they are kept for this session only (keep_settings); key 3
# changes the speed and Return writes PARLOR.CFG (keep_saved, peek k_nodrive).
waittitle 900
title 3600
waittext ParlorNet: connect 600
type S
waittext ParlorNet settings 300
wait 60
snap keep_settings
peek k_nodrive cfg_acia cfg_baud
type 3
wait 30
key ret
waittext ParlorNet: connect 2400
wait 300
type S
waittext ParlorNet settings 300
wait 60
snap keep_saved
peek k_nodrive cfg_acia cfg_baud
exit
