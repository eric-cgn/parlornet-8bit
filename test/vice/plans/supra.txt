# The Supra 2400 compatible modem, VICE's user port RS-232 (run.sh with IFACE=user and BAUD=2400): key 1
# makes the modem Supra 2400 compatible, at 2400 (and that is the speed); then the console, the
# dialer, signing up, the lobby and Commodore-Q.
waittitle 6000
title 3600
waittext ParlorNet: connect 600
type S
waittext ParlorNet settings 300
type 1
wait 30
snap supra_settings_set
key ret
waittext ParlorNet: connect 2400
snap supra_start
type C
waittext Line: 2400
type AT\r
waittext OK 300
snap supra_console
oa Q
waittext ParlorNet: connect 600
type P
waittext Name: 2400
snap supra_dialed
type new\r
waittext want 300
type sue\r
waittext password 600
type Swordfish 1997\r
waittext again 600
type Swordfish 1997\r
wait 900
snap supra_lobby
oa Q
waittext ParlorNet: connect 2400
snap supra_hungup
exit
