# The console on the serial line (run.sh's IFACE and BAUD), through the WiFi-modem stand-in: AT gets OK,
# ATDT connects, and C=-P goes into ParlorNet mode on the line as it is, to the name prompt.
waittitle 6000
title 3600
waittext ParlorNet: connect 600
type C
waittext Line: 4800
type AT\r
waittext OK 300
snap console_at
type ATDT parlornet.org:6501\r
waittext CONNECT 600
snap console
oa P
waittext Name: 600
snap signin
dumptext signin
exit
