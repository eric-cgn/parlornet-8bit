# The SwiftLink compatible modem at 38400: key 3 takes the speed from 19200 to 38400 (swift38_settings), Return
# keeps it (PARLOR.CFG), and the settings show it again (swift38_saved, peek cfg_baud: 6).
waittitle 6000
title 3600
waittext ParlorNet: connect 600
type S
waittext ParlorNet settings 300
wait 60
type 3
wait 30
snap swift38_settings
key ret
waittext ParlorNet: connect 2400
wait 600
snap swift38_start
type S
waittext ParlorNet settings 300
wait 60
snap swift38_saved
peek cfg_slot cfg_baud cfg_acia
exit
